/**
 * E12 shared helpers. Pricing, Luna cost, quantiles and jsonl come from the
 * E4 lib; the spend ledger is E4's Spend class rebased on this folder (the E4
 * ledger lives in e4-emoji/results and must not be written from here).
 *
 * Hard cap for everything in this folder: $0.40, including an estimate for
 * every speculative call that is aborted before its usage is known.
 */
import { mkdir, readFile, writeFile } from "node:fs/promises";
import { dirname, resolve } from "node:path";
import type { CallTrace } from "../../../src/types.ts";
import { lunaCost, price } from "../e4-emoji/lib.ts";

export { appendJsonl, quantile, round } from "../e4-emoji/lib.ts";
export { lunaCost, price };

export const here = dirname(new URL(import.meta.url).pathname);
export const resultsDir = resolve(here, "results");
export const CAP_USD = 0.4;
const RESERVE_USD = 0.012;
/** Charged for a Luna call we abort (no usage comes back): a full emoji call. */
export const ABORTED_EMOJI_ESTIMATE_USD = 0.00013;

type LedgerEntry = {
	script: string;
	label: string;
	startedAt: string;
	finishedAt?: string;
	spentUsd: number;
	lunaCalls: number;
	jevCalls: number;
	abortedEstimates: number;
	stoppedBy?: string;
};
const ledgerPath = resolve(resultsDir, "ledger.json");
async function readLedger(): Promise<LedgerEntry[]> {
	try {
		return JSON.parse(await readFile(ledgerPath, "utf8")) as LedgerEntry[];
	} catch {
		return [];
	}
}

export class Spend {
	prior = 0;
	spent = 0;
	lunaCalls = 0;
	jevCalls = 0;
	abortedEstimates = 0;
	skipped = 0;
	stoppedBy: string | undefined;
	private entry: LedgerEntry;
	private constructor(
		script: string,
		label: string,
		private readonly scriptCap: number,
	) {
		this.entry = {
			script,
			label,
			startedAt: new Date().toISOString(),
			spentUsd: 0,
			lunaCalls: 0,
			jevCalls: 0,
			abortedEstimates: 0,
		};
	}
	static async open(script: string, label: string, scriptCap: number) {
		await mkdir(resultsDir, { recursive: true });
		const spend = new Spend(script, label, scriptCap);
		spend.prior = (await readLedger()).reduce(
			(sum, entry) => sum + entry.spentUsd,
			0,
		);
		console.log(
			`[spend] prior ledger total $${spend.prior.toFixed(4)}; script cap $${scriptCap}; global cap $${CAP_USD}`,
		);
		return spend;
	}
	get total() {
		return this.prior + this.spent;
	}
	/** True when one more unit of about `estimate` USD still fits both caps. */
	allows(estimate: number) {
		const ok =
			this.prior + this.spent + estimate + RESERVE_USD <= CAP_USD &&
			this.spent + estimate <= this.scriptCap;
		if (!ok) this.skipped++;
		return ok;
	}
	add(usd: number, executor: "Luna" | "jev") {
		this.spent += usd;
		if (executor === "Luna") this.lunaCalls++;
		else this.jevCalls++;
	}
	addAbortedEstimate() {
		this.spent += ABORTED_EMOJI_ESTIMATE_USD;
		this.abortedEstimates++;
	}
	async close(stoppedBy?: string) {
		this.stoppedBy = stoppedBy;
		const ledger = await readLedger();
		ledger.push({
			...this.entry,
			finishedAt: new Date().toISOString(),
			spentUsd: this.spent,
			lunaCalls: this.lunaCalls,
			jevCalls: this.jevCalls,
			abortedEstimates: this.abortedEstimates,
			...(stoppedBy ? { stoppedBy } : {}),
		});
		await writeFile(ledgerPath, `${JSON.stringify(ledger, null, 2)}\n`);
		console.log(
			`[spend] this run $${this.spent.toFixed(5)} (${this.lunaCalls} Luna, ${this.jevCalls} jev, ${this.abortedEstimates} aborted-call estimates, ${this.skipped} skipped by cap); ledger total $${(this.prior + this.spent).toFixed(5)}`,
		);
	}
}

/** Cost of one traced call, as waterfall.ts usageOf. */
export function callCost(call: CallTrace) {
	if (call.executor === "TypeSafe") {
		const usage = (call.output as { usage?: { input_tokens?: number } })
			?.usage;
		const input = usage?.input_tokens ?? 0;
		return {
			executor: "jev" as const,
			inputTokens: input,
			cachedTokens: 0,
			outputTokens: 0,
			costUsd: input * price.jevInput,
			hasUsage: usage !== undefined,
		};
	}
	const metadata = call.metadata as
		| { usage?: Parameters<typeof lunaCost>[0] }
		| undefined;
	return {
		executor: "Luna" as const,
		...lunaCost(metadata?.usage ?? null),
		hasUsage: !!metadata?.usage,
	};
}

/** Billing/auth failures stop the run at once (no retry loops). */
export function isBillingOrAuth(message: string) {
	return /\b(402|401|403)\b|insufficient|quota|billing|payment|credits|unauthori[sz]ed|invalid api key|incorrect api key/iu.test(
		message,
	);
}
